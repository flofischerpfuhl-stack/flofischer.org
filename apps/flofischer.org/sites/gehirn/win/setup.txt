# Fernzugang fuer Claude einrichten: SSH-Server, Schluessel, Firewall, kein Standby, Tailscale.
# Aufruf in einem Administrator-Terminal:  irm https://gehirn.flofischer.org/win/setup.txt | iex
$ErrorActionPreference = 'Stop'
Write-Host '1/5 OpenSSH-Server installieren (dauert etwas) ...' -ForegroundColor Cyan
Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0 | Out-Null
Set-Service sshd -StartupType Automatic
Start-Service sshd
Write-Host '2/5 Schluessel von Laptop und PC freischalten ...' -ForegroundColor Cyan
$keyFile = 'C:\ProgramData\ssh\administrators_authorized_keys'
Set-Content -Path $keyFile -Encoding ascii -Value @(
  'ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAINYoVIIv5s2DpXo9vkE5XkMZdNoqfHNAylfm3jI74o5R flo@thinkpad-flo',
  'ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIBsIE8n6sKOrc0kGGWCzrYPSXFxJ2AEeLAeqGeahaDzR flo-cachyos'
)
icacls $keyFile /inheritance:r /grant '*S-1-5-32-544:F' /grant '*S-1-5-18:F' | Out-Null
New-ItemProperty -Path HKLM:\SOFTWARE\OpenSSH -Name DefaultShell -Value 'C:\Windows\System32\WindowsPowerShell\v1.0\powershell.exe' -PropertyType String -Force | Out-Null
Write-Host '3/5 Firewall fuer SSH oeffnen ...' -ForegroundColor Cyan
$rule = Get-NetFirewallRule -Name OpenSSH-Server-In-TCP -ErrorAction SilentlyContinue
if ($rule) { $rule | Set-NetFirewallRule -Enabled True -Profile Any }
else { New-NetFirewallRule -Name OpenSSH-Server-In-TCP -DisplayName 'OpenSSH Server (sshd)' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 22 -Profile Any | Out-Null }
Write-Host '4/5 Standby und Ruhezustand am Netzteil aus ...' -ForegroundColor Cyan
powercfg /change standby-timeout-ac 0
powercfg /change hibernate-timeout-ac 0
Write-Host '5/5 Tailscale installieren ...' -ForegroundColor Cyan
$msi = Join-Path $env:TEMP 'tailscale-setup.msi'
Invoke-WebRequest -UseBasicParsing 'https://pkgs.tailscale.com/stable/tailscale-setup-latest-amd64.msi' -OutFile $msi
Start-Process msiexec.exe -Wait -ArgumentList @('/i', "`"$msi`"", '/qn', 'TS_UNATTENDEDMODE=always')
Start-Process 'C:\Program Files\Tailscale\tailscale-ipn.exe'
Write-Host ''
Write-Host 'Fertig. Jetzt unten rechts im Tray auf das Tailscale-Symbol klicken -> "Log in" und mit flofischer.pfuhl@ anmelden.' -ForegroundColor Green
Write-Host "Rechnername: $env:COMPUTERNAME" -ForegroundColor Green
